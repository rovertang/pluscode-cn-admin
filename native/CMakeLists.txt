cmake_minimum_required(VERSION 3.26)
project(pluscode_admin VERSION 1.0.0 LANGUAGES C CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use a separate temporary build directory")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
option(PCAD_JNI "Build Java/Kotlin JNI entry points" ON)
option(PCAD_TESTS "Build native tests" ON)
option(PCAD_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
if(PCAD_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include(cmake/dependencies.cmake)
find_package(Threads REQUIRED)
add_library(pluscode_admin SHARED src/index.cpp src/c_api.cpp)
target_compile_definitions(pluscode_admin PRIVATE PCAD_BUILDING)
target_include_directories(pluscode_admin PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>" PRIVATE "${PCAD_JSON}")
target_link_libraries(pluscode_admin PRIVATE pcad_sqlite pcad_olc liblzma zlibstatic Threads::Threads ${CMAKE_DL_LIBS})
target_include_directories(pluscode_admin PRIVATE "${PCAD_ZLIB}" "${CMAKE_BINARY_DIR}/_deps/zlib-build" "${PCAD_XZ}/src/liblzma/api")
target_compile_options(pluscode_admin PRIVATE -Wall -Wextra -Wpedantic)
if(UNIX)
  target_link_options(pluscode_admin PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")
  target_link_options(pluscode_admin PRIVATE -Wl,-z,defs)
  set_property(TARGET pluscode_admin APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")
endif()
if(ANDROID)
  target_link_options(pluscode_admin PRIVATE -Wl,-z,max-page-size=16384)
endif()
if(PCAD_JNI)
  if(NOT ANDROID)
    find_path(PCAD_JNI_INCLUDE jni.h HINTS "$ENV{JAVA_HOME}/include" REQUIRED)
    find_path(PCAD_JNI_PLATFORM jni_md.h HINTS "${PCAD_JNI_INCLUDE}/linux" "${PCAD_JNI_INCLUDE}/darwin" "${PCAD_JNI_INCLUDE}/win32" REQUIRED)
    target_include_directories(pluscode_admin PRIVATE "${PCAD_JNI_INCLUDE}" "${PCAD_JNI_PLATFORM}")
  endif()
  target_sources(pluscode_admin PRIVATE src/jni.cpp)
endif()
add_executable(pcad-query examples/query.cpp)
target_link_libraries(pcad-query PRIVATE pluscode_admin)
set_target_properties(pcad-query PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
if(PCAD_TESTS)
  enable_testing()
  add_executable(pcad-tests tests/native_tests.cpp)
  target_link_libraries(pcad-tests PRIVATE pluscode_admin Threads::Threads)
  target_include_directories(pcad-tests PRIVATE "${PCAD_JSON}")
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(pcad-memory-probe tests/memory_probe.cpp)
    target_link_libraries(pcad-memory-probe PRIVATE pluscode_admin)
  endif()
  set(PCAD_TEST_DB "" CACHE FILEPATH "V2 database for native integration tests")
  if(PCAD_TEST_DB)
    add_test(NAME native-contract COMMAND pcad-tests "${PCAD_TEST_DB}")
  endif()
endif()
install(TARGETS pluscode_admin pcad-query LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
